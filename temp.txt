model InventoryMovement {
    id String @id @default(cuid())

    createdAt DateTime @default(now())
    updatedAt DateTime @updatedAt

    inventoryItemId String
    inventoryItem   InventoryItem @relation(fields: [inventoryItemId], references: [id], onDelete: Restrict)

    transactionType InventoryTransactionType
    status          InventoryMovementStatus  @default(PENDING)

    // ==========================================================================
    // FROM
    // ==========================================================================

    fromStorageLocationId String?
    fromStorageUnit       StorageLocation? @relation("InventoryFromStorage", fields: [fromStorageLocationId], references: [id], onDelete: SetNull)

    // ==========================================================================
    // TO
    // ==========================================================================

    toStorageLocationId String?

    toStorageUnit StorageLocation? @relation("InventoryToStorage", fields: [toStorageLocationId], references: [id], onDelete: SetNull)

    quantity Decimal @db.Decimal(12, 4)

    referenceNumber String? @db.VarChar(100)

    notes String? @db.Text

    metadata Json?

    performedByUserId String?
    performedBy       User?   @relation("MovementPerformedBy", fields: [performedByUserId], references: [id], onDelete: SetNull)

    startedAt   DateTime?
    completedAt DateTime?

    @@index([inventoryItemId])
    @@index([transactionType])
    @@index([status])
    @@index([completedAt])
    @@index([startedAt])
    @@index([createdAt])
    @@index([referenceNumber])
    @@index([performedByUserId])
    @@map("inventory_movement")
}
